# MEMORY, not CPU, decides whether vocal mode is usable on App Service:
#   musicgen-melody = 1.55 B params = 6.2 GB fp32 / 3.1 GB bf16 of weights that
#   every decode step touches.  B2 has 3.5 GB RAM → the model can never be
#   resident → it pages from disk on all ~1 500 steps → "a day" per track.
#   Minimum for vocal mode: 8 GB (P1v3) + ARK_DTYPE=bfloat16; comfortable: 16 GB.
#   Text mode (musicgen-small, 1.2 GB) is fine on B2.
#   Add to the settings block when on a ≥ 8 GB plan:   ARK_DTYPE=bfloat16
# ARK_DATA_DIR: persistent root for the job DB + generated tracks + uploads.
# MUST be under /home and OUTSIDE wwwroot so a `--clean` deploy can't wipe it.
# startup.sh also defaults it to /home/data, so this is belt-and-suspenders.
# Optional: ARK_MAX_JOB_ATTEMPTS (default 3) caps auto-resume retries per job.

> az webapp config appsettings set \

==== run locally on M5 mac (Apple Silicon)

# NOTE: differs from the Azure notes above on purpose:
#   - DO NOT use the CPU-only wheel index (--index-url .../whl/cpu). On Apple
#     Silicon the default PyPI wheel is the native arm64 build and includes the
#     MPS (Metal) backend. The cpu index gives a slower, MPS-less build.
#   - ffmpeg comes from Homebrew locally (startup.sh's static-binary install is
#     Azure-only). audio_io.find_ffmpeg() picks it up off PATH automatically.

# ── ffmpeg (needed for MP3 export) ──────────────────────────────────
> brew install ffmpeg          # skip if `ffmpeg -version` already works

# ── one-time setup ──────────────────────────────────────────────────
> python3 -m venv .venv
> source .venv/bin/activate
> pip install --upgrade pip
> pip install torch torchaudio        # native arm64 wheel — NOT the cpu index
> pip install -r requirements.txt

# ── CLI ─────────────────────────────────────────────────────────────
# Activate the venv first in every new terminal:
> source .venv/bin/activate

> python main.py --help

# Default device is CPU — fastest here: MusicGen decodes one token at a time,
# so for musicgen-small MPS benchmarked ~2x SLOWER than a threaded CPU decode.
> python main.py "a cheerful whistling tune in C major" --duration 10 --output test.mp3

> python main.py "generate a typical ilayaraja song which reflects tamil" \
    --duration 10 --output test_i1.mp3

# Full manual control
> python main.py "jazz piano riff with walking bass" \
    --genre jazz --mood uplifting \
    --instruments piano,bass,drums \

# Vocal / melody source file (uses the larger musicgen-melody model)
> python main.py /path/to/melody.wav --genre pop --mood happy --output singalong.mp3
> python main.py isaiah_54_17.mp3 --genre pop --mood happy --output singalong.mp3

# ── vocal → music locally (the mode that produced ark_vocal_track.mp3) ─────
# The melody model is 1.55 B params = 6.2 GB in fp32.  On this 16 GB box with a
# VM + Teams + browser open that does NOT stay resident: the job that sat at
# "Generating accompaniment 1/2… 16%" for 6.5 h was paging weights from swap
# (17 GB swap in use).  bf16 halves it to 3.1 GB and is now the default on
# Apple Silicon (generator._select_dtype).  Measured, musicgen-melody, 30 tokens:
#   cpu/float32  → swapping, minutes per handful of tokens
#   cpu/bfloat16 → 0.05 s/token   (30 s clip ≈ 1.5 min incl. load)
#   mps/bfloat16 → 0.06 s/token
#   mps/float16  → 0.05 s/token   (fp16 T5 can overflow; prefer bf16)
# Free memory first (close the VM) or it will still swap.

> python vocalize.py isaiah_54_17.mp3 -o ark_vocal_track_v2.mp3
#   → ark_vocal_track_v2.mp3 (mix) + ark_vocal_track_v2_accompaniment.mp3
#   Output starts where the singing starts (this file has a 13 s silent
#   count-in) and covers the first 30 s of singing (ARK_VOCAL_MAX_SECONDS).

> ARK_DEVICE=mps python vocalize.py isaiah_54_17.mp3 -o ark_vocal_track_v2.mp3   # GPU
> ARK_DTYPE=float32 python vocalize.py …                                          # force fp32
> ARK_VOCAL_MAX_SECONDS=60 python vocalize.py …   # 2 × 30 s windows, chained by audio continuation

# ── tuning knobs (all optional) ─────────────────────────────────────
# This box: 4 performance + 6 efficiency cores. Default uses all 10; pinning
# near the P-core count avoids E-core memory-bandwidth contention:
> ARK_NUM_THREADS=4 python main.py "upbeat pop chorus" --duration 8 --output quick_test.mp3

# A/B the Metal GPU (worth trying on the bigger melody model; CPU wins on small).
# PYTORCH_ENABLE_MPS_FALLBACK is set by the code automatically for unimplemented ops:
> ARK_DEVICE=mps python main.py "energetic rock anthem" --genre rock --mood energetic --duration 20

# Force CPU regardless of ARK_DEVICE (matches the Azure path):
> ARK_FORCE_CPU=1 python main.py "lively folk tune with a bouncy rhythm"
